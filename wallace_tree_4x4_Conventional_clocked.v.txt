//=========================================================
// Full Adder -combinational building block 
//=========================================================
module full_adder (
    input  a,
    input  b,
    input  cin,
    output reg sum,
    output reg cout
);
    always @(*) begin
        sum  = a ^ b ^ cin;
        cout = (a & b) | (b & cin) | (a & cin);
    end
endmodule
 
 
//=========================================================
// Half Adder (combinational building block 
//=========================================================
module half_adder (
    input  a,
    input  b,
    output reg sum,
    output reg carry
);
    always @(*) begin
        sum   = a ^ b;
        carry = a & b;
    end
endmodule
 
 
module Wallace_summer (
    input         clk,
    input         rst,     // synchronous, active-high
    input  [3:0]  A,
    input  [3:0]  B,
    output [7:0]  P
);
 
    // ---------------- Stage 1: Input capture register ----------------
    reg [3:0] A_reg, B_reg;
    always @(posedge clk) begin
        if (rst) begin
            A_reg <= 4'b0;
            B_reg <= 4'b0;
        end else begin
            A_reg <= A;
            B_reg <= B;
        end
    end
 
    // ---------------- Stage 2: Combinational Wallace tree ----------------
    wire p00,p01,p02,p03;
    wire p10,p11,p12,p13;
    wire p20,p21,p22,p23;
    wire p30,p31,p32,p33;
 
    and G0  (p00,A_reg[0],B_reg[0]);
    and G1  (p01,A_reg[0],B_reg[1]);
    and G2  (p02,A_reg[0],B_reg[2]);
    and G3  (p03,A_reg[0],B_reg[3]);
    and G4  (p10,A_reg[1],B_reg[0]);
    and G5  (p11,A_reg[1],B_reg[1]);
    and G6  (p12,A_reg[1],B_reg[2]);
    and G7  (p13,A_reg[1],B_reg[3]);
    and G8  (p20,A_reg[2],B_reg[0]);
    and G9  (p21,A_reg[2],B_reg[1]);
    and G10 (p22,A_reg[2],B_reg[2]);
    and G11 (p23,A_reg[2],B_reg[3]);
    and G12 (p30,A_reg[3],B_reg[0]);
    and G13 (p31,A_reg[3],B_reg[1]);
    and G14 (p32,A_reg[3],B_reg[2]);
    and G15 (p33,A_reg[3],B_reg[3]);
 
    wire c1,c2,c3,c4,c5,c6,c7,c8,c9,c10,c11;
    wire s2,s3,s4,s5,s6,s7,s8,s9,s10,s11;
    wire [7:0] P_comb;
 
    assign P_comb[0] = p00;
    half_adder HA1(p10,p01,P_comb[1],c1);
    full_adder FA1(p20,p11,p02,s2,c2);
    half_adder HA2(s2,c1,P_comb[2],c7);
    full_adder FA2(p30,p21,p12,s3,c3);
    full_adder FA3(s3,c2,p03,s8,c8);
    assign P_comb[3] = s8;
    full_adder FA4(p31,p22,p13,s4,c4);
    full_adder FA5(s4,c7,c8,P_comb[4],c9);
    full_adder FA6(p32,p23,c3,s5,c5);
    full_adder FA7(s5,c5,c9,P_comb[5],c10);
    half_adder HA3(p33,c4,s6,c6);
    full_adder FA8(s6,c6,c10,P_comb[6],c11);
    assign P_comb[7] = c11;
 
    // ---------------- Stage 3: Output capture register ----------------
    reg [7:0] P_reg;
    always @(posedge clk) begin
        if (rst)
            P_reg <= 8'b0;
        else
            P_reg <= P_comb;
    end
 
    assign P = P_reg;
 
endmodule
 
 
//=========================================================
// Testbench (clocked stimulus)
//=========================================================
module Wallace_summer_tb();
    reg         clk;
    reg         rst;
    reg  [3:0]  A;
    reg  [3:0]  B;
    wire [7:0]  P;
    integer i, j;
 
    // Instantiate DUT
    Wallace_summer U1 (
        .clk(clk),
        .rst(rst),
        .A(A),
        .B(B),
        .P(P)
    );
 
    // 10 ns period clock (100 MHz)
    initial clk = 0;
    always #5 clk = ~clk;
 
    initial begin
        $display(" A  B | Product");
 
        // reset for a couple of cycles
        rst = 1;
        A   = 0;
        B   = 0;
        @(posedge clk);
        @(posedge clk);
        rst = 0;
 
        for (i = 0; i < 16; i = i + 1) begin
            for (j = 0; j < 16; j = j + 1) begin
                A = i;
                B = j;
                @(posedge clk);  // edge 1: A,B captured into A_reg,B_reg
                @(posedge clk);  // edge 2: P_comb captured into P_reg -> P valid
                $display("%2d %2d | %3d", A, B, P);
            end
        end
 
        $finish;
    end
endmodule
